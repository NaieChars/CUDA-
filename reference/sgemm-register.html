Register GEMM：线程内的二维复用

返回练习 05 · Shared 版速查

固定配置与三个独立分工

BX=BY=16，BM=BN=32，BK=16，TM=TN=2。256 个线程覆盖 1024 个输出；动态 shared 为 (BM×BK+BK×BN)×4=4096 字节。以下是本课选择的一种合法分工，配套加载、累加与写回必须一致。

用途规则
输出归属r_i=ty+i×BY，c_j=tx+j×BX；C[by×BM+r_i,bx×BN+c_j]
shared 加载归属tid=ty×BX+tx，q=tid+s×(BX×BY)，A/B 各自遍历
A 槽位 qtile 行=q/BK，K 列=q%BK；global 行宽 K
B 槽位 qK 行=q/BN，tile 列=q%BN；global 行宽 N
一个 e 的私有片段a_frag[i]=As[r_i,e]；b_frag[j]=Bs[e,c_j]
一个 e 的外积acc[i][j]+=a_frag[i]×b_frag[j]

acc[2][2] 是逻辑分组，所对应的输出行列在本例中间隔 16。输入越界补零；每段先发布 shared，再计算，读完后才能覆盖；四个输出各自判断合法性。线程自己的寄存器片段不需要额外同步。

独立完成后展开核对

查看 sgemm_register 参考实现
__global__ void sgemm_register(const float* __restrict__ a,
                               const float* __restrict__ b,
                               float* __restrict__ c,
                               size_t m, size_t n, size_t k)
{
    constexpr int BX = sgemm_config::kBlockX;
    constexpr int BY = sgemm_config::kBlockY;
    constexpr int BM = sgemm_config::kRegisterTileM;
    constexpr int BN = sgemm_config::kRegisterTileN;
    constexpr int BK = sgemm_config::kRegisterTileK;
    constexpr int TM = sgemm_config::kThreadTileM;
    constexpr int TN = sgemm_config::kThreadTileN;
    constexpr int NT = BX * BY;

    const int tx = threadIdx.x, ty = threadIdx.y;
    const int tid = ty * BX + tx;
    const size_t block_row = static_cast<size_t>(blockIdx.y) * BM;
    const size_t block_col = static_cast<size_t>(blockIdx.x) * BN;

    extern __shared__ float smem[];
    float* as = smem;
    float* bs = smem + BM * BK;
    float acc[TM][TN] = {};

    for (size_t k0 = 0; k0 < k; k0 += BK)
    {
        for (int q = tid; q < BM * BK; q += NT)
        {
            const size_t row = block_row + q / BK;
            const size_t p = k0 + q % BK;
            as[q] = (row < m && p < k)
                ? a[row * k + p] : 0.0f;
        }
        for (int q = tid; q < BK * BN; q += NT)
        {
            const size_t p = k0 + q / BN;
            const size_t col = block_col + q % BN;
            bs[q] = (p < k && col < n)
                ? b[p * n + col] : 0.0f;
        }
        __syncthreads();

        for (int e = 0; e < BK; ++e)
        {
            float a_frag[TM], b_frag[TN];
            #pragma unroll
            for (int i = 0; i < TM; ++i)
                a_frag[i] = as[(ty + i * BY) * BK + e];
            #pragma unroll
            for (int j = 0; j < TN; ++j)
                b_frag[j] = bs[e * BN + tx + j * BX];

            #pragma unroll
            for (int i = 0; i < TM; ++i)
            {
                #pragma unroll
                for (int j = 0; j < TN; ++j)
                    acc[i][j] += a_frag[i] * b_frag[j];
            }
        }
        __syncthreads();
    }

    #pragma unroll
    for (int i = 0; i < TM; ++i)
    {
        const size_t row = block_row + ty + i * BY;
        #pragma unroll
        for (int j = 0; j < TN; ++j)
        {
            const size_t col = block_col + tx + j * BX;
            if (row < m && col < n)
                c[row * n + col] = acc[i][j];
        }
    }
}

核函数仍由你手写；完成后才启用 kRegisterImplemented。#pragma unroll 表达展开意图，不构成物理寄存器分配或无溢出的证据。

四个容易混淆的地方

block 输出起点乘 BM/BN，不是 blockDim。As 的行宽 BK，Bs 的行宽 BN。每个 shared tile 有 512 个值而线程只有 256 个，必须加载第二轮 q。输出为分散的四个坐标，写回不能复用首个输出的边界条件。

准备验证的假设

每个 e 内，一线程读取 TM+TN=4 个 shared 标量，完成 TM×TN=4 次乘加；四个独立输出线程按上一版写法需要合计 8 次读取。这个源码层面的 shared 读取计数减少一半，不保证 profiler 请求或服务周期减半。四个累加器提供独立依赖链，但实际调度收益仍待测。

本版还把 block 输出从 16×16 扩到 32×32。对同一 32×32 输出、完整 BK=16，前版四个 block 共加载 2048 个 global 输入标量，本版一个 block 加载 1024 个。这个收益来自更大的输出 tile；三版成绩不能隔离纯粹的线程内寄存器复用效果。代价与待查项包括寄存器数量、local/spill、shared 占用和问题总 block 数。
